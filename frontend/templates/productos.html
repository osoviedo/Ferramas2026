// Conversión de precios visibles según la moneda seleccionada. // Los valores vienen del backend (DivisaService -> mindicador.cl / Banco Central). // El carrito y los totales se mantienen SIEMPRE en CLP (data-precio-clp). const DIVISAS = { dolar: {{ divisas.dolar | tojson }}, euro: {{ divisas.euro | tojson }} }; const CONVERT_TO = "{{ convert_to }}"; function formatearPrecios() { document.querySelectorAll(".producto-precio").forEach(function (el) { const clp = parseFloat(el.dataset.precioClp); if (isNaN(clp)) return; if (CONVERT_TO === "USD" && DIVISAS.dolar) { const v = clp / DIVISAS.dolar; el.textContent = "US$ " + v.toLocaleString("es-CL", { minimumFractionDigits: 2, maximumFractionDigits: 2 }); } else if (CONVERT_TO === "EUR" && DIVISAS.euro) { const v = clp / DIVISAS.euro; el.textContent = "\u20AC " + v.toLocaleString("es-CL", { minimumFractionDigits: 2, maximumFractionDigits: 2 }); } else { el.textContent = "$" + clp.toLocaleString("es-CL"); } }); } formatearPrecios(); {% endblock %}
